% figures/tikz/hardware_blocks.tex --- what connects to the Pi, by bus, plus power.
% Pins follow vision_stack/src/params.py and src/peripherals/drive.py (BCM numbering).
% \input{figures/tikz/hardware_blocks} inside a figure environment.
\begin{tikzpicture}[
    part/.style  = {draw, rounded corners=2pt, fill=black!4, align=center,
                    font=\small, minimum height=9mm, minimum width=24mm},
    pi/.style    = {draw, very thick, rounded corners=3pt, fill=blue!10,
                    align=center, minimum width=38mm, minimum height=34mm},
    power/.style = {part, fill=red!8},
    sig/.style   = {-{Latex[length=1.8mm]}, thick},
    bus/.style   = {{Latex[length=1.8mm]}-{Latex[length=1.8mm]}, thick},
    pwr/.style   = {-{Latex[length=1.8mm]}, thick, red!60!black},
    lbl/.style   = {font=\scriptsize, align=center, fill=white, inner sep=1pt},
  ]
  % The Pi
  \node[pi] (pi) at (0,0) {\textbf{Raspberry Pi}\\\textbf{Zero 2 W}\\[2pt]
                           {\footnotesize all software}};

  % Left: sensors
  \node[part] (adc) at (-5.6, 1.5)  {ADS1115\\{\scriptsize ADC, 0x48}};
  \node[part] (imu) at (-5.6, 0.1)  {MPU-6050\\{\scriptsize IMU, 0x68}};
  \node[part] (cam) at (-5.6,-1.4)  {IMX290 camera};

  \draw[bus] (adc.east) -- node[lbl, above=1pt] {I\textsuperscript{2}C, 400\,kHz}
             (adc.east -| pi.west);
  \draw[bus] (imu.east) -- (imu.east -| pi.west);
  \draw[sig] (cam.east) -- node[lbl, above=1pt] {CSI-2} (cam.east -| pi.west);

  % Right: actuation
  \node[part] (drv) at (5.7, 0.9)  {TB6612FNG\\{\scriptsize H-bridge}};
  \node[part, minimum width=17mm] (mot) at (8.55, 0.9) {2$\times$ N20\\motors};

  \draw[sig] (drv.west -| pi.east) -- node[lbl, above=1pt] {PWM 12/13\\dir 22, 27, 25, 24}
             (drv.west);
  \draw[sig] (drv) -- (mot);
  \draw[sig] (mot.south) |- node[lbl, pos=0.75, below=1pt] {encoders 19/16, 20/21}
             ([yshift=-8mm]pi.east);

  % Bottom: user interface
  \node[part] (disp) at (-1.3,-3.2) {TM1637\\{\scriptsize display}};
  \node[part] (btn)  at ( 1.6,-3.2) {Start\\{\scriptsize button}};
  \draw[sig] (disp.north |- pi.south) -- node[lbl, left=1pt] {GPIO 5/6} (disp.north);
  \draw[sig] (btn.north) -- node[lbl, right=1pt] {GPIO 17} (btn.north |- pi.south);

  % Top: power
  \node[power] (bat)  at (5.7, 3.4) {3S 18650 pack\\{\scriptsize 9.0--12.6\,V}};
  \node[power] (buck) at (0.0, 3.4) {Buck converter};
  \draw[pwr] (bat) -- (buck);
  \draw[pwr] (buck) -- node[lbl, right=1pt] {5\,V} (pi.north);
  \draw[pwr] (bat) -- node[lbl, right=1pt] {VM (battery)} (drv);
  \draw[pwr, dashed] (bat.north) -- ++(0,0.45) -| node[lbl, pos=0.25, above=1pt]
        {battery sense, 4:1 divider} (adc.north);
\end{tikzpicture}
